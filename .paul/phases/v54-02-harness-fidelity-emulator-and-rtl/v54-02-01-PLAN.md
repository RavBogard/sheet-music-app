    describe('emulator canary (v54-02-01) — proves Firebase Local Emulator Suite wiring', () => {
        let db: import('firebase-admin/firestore').Firestore
        beforeAll(async () => {
            const { initializeApp, getApps } = await import('firebase-admin/app')
            const { getFirestore } = await import('firebase-admin/firestore')
            if (getApps().length === 0) {
                initializeApp({ projectId: 'demo-v54-02-canary' })
            }
            db = getFirestore()
            // Sanity: env var must be set by emulators:exec wrapper
            expect(process.env.FIRESTORE_EMULATOR_HOST).toBeTruthy()
        })
        it('round-trips a doc through the emulator', async () => {
            const ref = db.doc('canary/v54-02-01')
            await ref.set({ ok: true, ts: 5000 })
            const snap = await ref.get()
            expect(snap.exists).toBe(true)
            expect(snap.data()).toMatchObject({ ok: true, ts: 5000 })
        })
    })
    ```

        - uses: actions/setup-node@v4

        - run: npm ci
        - name: Cache Firebase emulator binaries
          uses: actions/cache@v4
          with:
            path: ~/.cache/firebase/emulators
            key: ${{ runner.os }}-firebase-emulators-${{ hashFiles('firebase.json') }}
        - name: Run emulator tests
          run: npm run test:emulator:ci
          env:
            SKIP_ENV_VALIDATION: '1'
    ```
