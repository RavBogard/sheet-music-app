    ```typescript
      async (ctx) => {
        const limited = await checkRateLimit(ctx.req, 'api')
        if (limited) return limited
        // ... existing logic using ctx.auth and ctx.body
      },
      { schema }  // or { role: 'band_leader', schema } if role-restricted
    )
    ```
