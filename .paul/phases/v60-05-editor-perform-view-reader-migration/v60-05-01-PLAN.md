<objective>
## Goal
Extract the 3-branch hydration-aware reader logic embedded in `use-setlist-performance.ts:125-130` into a shared, unit-tested `getTracksForSetlistClient(dexieTracks, setlistData)` helper. Refactor the perf-view hook to call it. Editor side requires NO migration (SetlistGrid already reads from Dexie via `useLiveQuery`, with `initialTracks` flowing through the SSR fetcher that v60-04-01 already routed through the server helper).

## Purpose
This closes the symmetry between server (`getTracksForSetlist` at @/lib/server-tracks) and client (`getTracksForSetlistClient` at @/lib/client-tracks) hydration-aware reader contracts. v6.0's "single source of truth" promise is enforced at the type/API level, not just at the engine level. v60-06 (dashboard reader migration) and v60-08 (legacy-branch cleanup — drops the unhydrated fallback branch from both helpers in lockstep) consume this helper as the canonical client-side reader. Live-worship perf-view rendering is sensitive — bundling the dual-read logic inside a hook makes the contract impossible to test in isolation; extracting it lets us prove the 3 branches behave correctly without booting the full hook.

## Output
- NEW `src/lib/client-tracks.ts` — exports `getTracksForSetlistClient(dexieTracks, setlistData)` with the same 3-branch behavior currently inlined at use-setlist-performance.ts:125-130.

- `src/hooks/use-setlist-performance.ts` — `useMemo` ternary replaced by helper call; same return type and reactivity.
- Single combined commit per v60-01..v60-04 precedent.

      import { describe, it, expect } from 'vitest'
      import { getTracksForSetlistClient } from '@/lib/client-tracks'

      describe('v60-05-01 getTracksForSetlistClient (pure function)', () => {
          // Branch 1: hydrated → Dexie authoritative, embedded ignored
          it('hydrated branch: returns dexieTracks; ignores stale embedded tracks', () => {
              const dexie = [{ id: 't1', setlistId: 's1', order: 0, title: 'Adon Olam' }]
              const setlist = { hydrated: true, tracks: [{ title: 'STALE' }] }
              const out = getTracksForSetlistClient(dexie as any, setlist)
              expect(out).toHaveLength(1)
              expect(out[0].title).toBe('Adon Olam')
          })

          // Branch 2: Dexie-has-data pre-hydration (snapshot-listener delivered)
          it('Dexie-has-data pre-hydration: returns dexieTracks (NOT embedded fallback)', () => {
              const dexie = [{ id: 't1', setlistId: 's1', order: 0, title: 'Snapshot Delivered' }]
              const setlist = { hydrated: false, tracks: [{ title: 'EMBEDDED' }] }
              const out = getTracksForSetlistClient(dexie as any, setlist)
              expect(out[0].title).toBe('Snapshot Delivered')
          })

          // Branch 3: unhydrated + empty Dexie → embedded fallback
          it('unhydrated + empty Dexie: returns embedded tracks', () => {
              const dexie: any[] = []
              const setlist = { hydrated: false, tracks: [{ title: 'Embedded 1' }, { title: 'Embedded 2' }] }
              const out = getTracksForSetlistClient(dexie, setlist)
              expect(out.map(t => t.title)).toEqual(['Embedded 1', 'Embedded 2'])
          })

          // Edge case: undefined dexieTracks (in-flight useLiveQuery)
          it('undefined dexieTracks (in-flight live query) + hydrated setlist → empty array', () => {
              const out = getTracksForSetlistClient(undefined, { hydrated: true, tracks: [] })
              expect(out).toEqual([])
          })

          // Edge case: undefined dexieTracks + unhydrated → embedded fallback
          it('undefined dexieTracks + unhydrated setlist → embedded tracks', () => {
              const out = getTracksForSetlistClient(undefined, {
                  hydrated: false,
                  tracks: [{ title: 'Legacy' }],
              })
              expect(out).toHaveLength(1)
              expect(out[0].title).toBe('Legacy')
          })

          // Edge case: null/undefined setlistData
          it('undefined setlistData → empty array', () => {
              expect(getTracksForSetlistClient([], undefined)).toEqual([])
              expect(getTracksForSetlistClient([], null)).toEqual([])
          })

          // Edge case: setlistData with no tracks field
          it('setlistData without tracks field + empty Dexie → empty array', () => {
              const out = getTracksForSetlistClient([], { hydrated: false })
              expect(out).toEqual([])
          })
      })
